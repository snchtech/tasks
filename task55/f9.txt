Так. Сам Ajax Callback краще трохи переробити, щоб він приймав формат через `x01` і залежно від нього викликав PDF або XLSX.

Тобто один callback `DOWNLOAD_REPORT`, а в ньому:

```plsql
declare
    l_export       apex_data_export.t_export;
    l_context      apex_exec.t_context;
    l_print_config apex_data_export.t_print_config;
    l_columns      apex_data_export.t_columns;
    l_report_id    number;
    l_format       varchar2(10);
begin
    l_format := upper(apex_application.g_x01);

    apex_data_export.add_column(
        p_columns => l_columns,
        p_name    => 'TASK_INFO',
        p_heading => 'Інформація'
    );

    apex_data_export.add_column(
        p_columns => l_columns,
        p_name    => 'TASK_DESCRIPTION_HTML',
        p_heading => 'Опис'
    );

    apex_data_export.add_column(
        p_columns => l_columns,
        p_name    => 'CREATOR_EXECUTOR_INFO',
        p_heading => 'Виконавець'
    );

    apex_data_export.add_column(
        p_columns => l_columns,
        p_name    => 'NOTE',
        p_heading => 'Примітка'
    );

    l_report_id :=
        apex_ir.get_last_viewed_report_id(
            p_page_id   => 15,
            p_region_id => 397351893134453127
        );

    l_context :=
        apex_region.open_query_context(
            p_page_id      => 15,
            p_region_id    => 397351893134453127,
            p_component_id => l_report_id,
            p_view_mode    => 'REPORT'
        );

    if l_format = 'PDF' then

        l_print_config :=
            apex_data_export.get_print_config(
                p_paper_size  => apex_data_export.c_size_a4,
                p_orientation => apex_data_export.c_orientation_portrait
            );

        l_export :=
            apex_data_export.export(
                p_context      => l_context,
                p_format       => apex_data_export.c_format_pdf,
                p_print_config => l_print_config,
                p_columns      => l_columns,
                p_page_header  => 'Звіт по задачах',
                p_file_name    => 'task_report'
            );

    elsif l_format = 'XLSX' then

        l_export :=
            apex_data_export.export(
                p_context   => l_context,
                p_format    => apex_data_export.c_format_xlsx,
                p_columns   => l_columns,
                p_file_name => 'task_report'
            );

    else
        raise_application_error(
            -20001,
            'Невідомий формат експорту: ' || l_format
        );
    end if;

    apex_exec.close(l_context);

    apex_data_export.download(
        p_export => l_export
    );

    apex_application.stop_apex_engine;

exception
    when apex_application.e_stop_apex_engine then
        null;

    when others then
        if apex_exec.is_open(l_context) then
            apex_exec.close(l_context);
        end if;

        raise;
end;
```

У JS ти вже передаєш:

```javascript
x01: format
```

Тобто для PDF:

```javascript
downloadReport("PDF");
```

а для Excel:

```javascript
downloadReport("XLSX");
```

І callback сам вирішує:

```plsql
PDF  -> apex_data_export.c_format_pdf
XLSX -> apex_data_export.c_format_xlsx
```

Для Excel `l_print_config` не потрібен взагалі. Він потрібен тільки для PDF, де є `A4`, `Portrait` тощо.

Тобто два окремих Ajax process тобі не потрібні — один `DOWNLOAD_REPORT` нормально обслуговує обидві кастомні кнопки.
